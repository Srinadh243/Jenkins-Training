pipeline {
    agent any // Runs this pipeline on any available executor node

    environment {
        APP_NAME = 'my-app'
    }

    stages {
        stage('Checkout') {
            steps {
                echo 'Fetching source code...'
                // If using Jenkinsfile from Git, SCM checkout happens automatically
            }
        }

        stage('Build') {
            steps {
                echo 'Building the application...'
                // Example for Node.js: sh 'npm install && npm run build'
                // Example for Java/Maven: sh './mvnw clean package'
            }
        }

        stage('Test') {
            steps {
                echo 'Running automated tests...'
                // Example: sh 'npm test' or sh './mvnw test'
            }
        }

        stage('Deploy') {
            steps {
                echo 'Deploying application to production...'
                // Example script to trigger deployment
            }
        }
    }

    post {
        always {
            echo 'Cleaning up workspace...'
        }
        success {
            echo 'Pipeline completed successfully!'
        }
        failure {
            echo 'Pipeline failed. Check the logs.'
        }
    }
}
